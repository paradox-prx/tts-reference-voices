DEFAULT_REF_AUDIO = "reference_voice.wav"#kejigan.wav"

@asynccontextmanager
async def lifespan(app: FastAPI):
    """服务生命周期管理：启动加载模型，关闭清理资源"""
    global global_qwen_tts_model
    
    try:
        print("✅ 开始加载Qwen3-TTS模型（仅加载一次，后续请求复用）...")
        start_load_time = time.perf_counter()  # 新增：模型加载计时开始
        
        global_qwen_tts_model = Qwen3TTSModel.from_pretrained(
            "data/Qwen3-TTS-12Hz-1.7B-Base",
            device_map="cuda:0",
            dtype=torch.bfloat16,
        )
        
        load_cost = time.perf_counter() - start_load_time  # 新增：计算模型加载耗时
        print(f"✅ Qwen3-TTS模型加载成功！加载耗时：{load_cost:.4f} 秒")
    except Exception as e:
        raise HTTPException(
            status_code=500,
            detail=f"模型加载失败：{str(e)}"
        )
    
    yield
    
    # 服务关闭：清理资源
#     global global_qwen_tts_model
    global_qwen_tts_model = None
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
    print("❌ 服务已关闭，模型资源已清理！")

app = FastAPI(

# 5. 定义请求体模型
# ======================================
class TTSRequest(BaseModel):
    text: str = Field(..., description="需要合成语音的文本内容")
    spk_id: int = Field(0, description="说话人ID（兼容原有接口，暂对应默认参考音频/文本）")
    speed: float = Field(1.0, description="语音语速（>0，1.0为默认语速）")
    volume: float = Field(1.0, description="语音音量（>0，1.0为默认音量）")
    sample_rate: int = Field(0, description="采样率（0为默认，使用模型返回的采样率）")
    save_path: str = Field("./output_voice_clone.wav", description="合成音频保存路径（wav格式）")
    ref_audio: Optional[str] = Field(None, description="自定义参考音频路径（可选，覆盖默认kejigan.wav）")
    ref_text: Optional[str] = Field(None, description="自定义参考文本（可选，覆盖默认参考文本）")

# ======================================
# 6. 核心接口：/paddlespeech/tts（新增全流程日志+耗时统计）
# ======================================
@app.post("/paddlespeech/tts", summary="语音合成接口（支持语音克隆，带耗时统计）")
async def tts_service(request: TTSRequest):
    """
    接收JSON请求，执行语音合成，返回全流程耗时统计
    """
    # ======================================
    # 新增：阶段1 - 接收请求，总耗时开始
    # ======================================
    total_start_time = time.perf_counter()
    request_text_summary = request.text[:50] + "..." if len(request.text) > 50 else request.text
    print(f"\n==================================================")
    print(f"📩 [{time.strftime('%Y-%m-%d %H:%M:%S')}] 接收新请求 - 文本摘要：{request_text_summary}")
    print(f"📩 保存路径：{request.save_path} | 语速：{request.speed} | 音量：{request.volume}")
    
    # 1. 校验模型是否已加载
    if global_qwen_tts_model is None:
        total_cost = time.perf_counter() - total_start_time
        print(f"❌ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 请求处理失败 - 模型未加载完成")
        print(f"❌ 总耗时：{total_cost:.4f} 秒")
        print(f"==================================================")
        raise HTTPException(
            status_code=500,
            detail="模型未加载完成，请稍后再试"
        )
    
    try:
        # 2. 配置参数
        ref_audio = request.ref_audio if request.ref_audio else DEFAULT_REF_AUDIO
        ref_text = request.ref_text if request.ref_text else DEFAULT_REF_TEXT
        save_path = request.save_path
        speed = request.speed
        volume = request.volume
        
        # 3. 校验参数合法性
        if speed <= 0 or volume <= 0:
            total_cost = time.perf_counter() - total_start_time
            print(f"❌ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 请求处理失败 - 语速/音量必须大于0")
            print(f"❌ 总耗时：{total_cost:.4f} 秒")
            print(f"==================================================")
            raise HTTPException(
                status_code=400,
                detail="语速（speed）和音量（volume）必须大于0"
            )
        
        # ======================================
        # 新增：阶段2 - 模型推理，开始计时
        # ======================================
        infer_start_time = time.perf_counter()
        print(f"⚙️ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 开始模型推理 - 参考音频：{ref_audio}")
        
        # 4. 模型推理生成语音
        wavs, sr = global_qwen_tts_model.generate_voice_clone(
            text=request.text,
            language=DEFAULT_LANGUAGE,
            ref_audio=ref_audio,
            ref_text=ref_text,
        )
        
        # 新增：计算模型推理耗时
        infer_cost = time.perf_counter() - infer_start_time
        print(f"✅ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 模型推理完成 - 耗时：{infer_cost:.4f} 秒")
        
        # ======================================
        # 新增：阶段3 - 音频处理，开始计时
        # ======================================
        audio_process_start_time = time.perf_counter()
        print(f"⚙️ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 开始音频处理 - 调整语速/音量")
        
        # 5. 调整语速、音量
        wav_data = wavs[0]
        if speed != 1.0:
            wav_data = librosa.effects.time_stretch(wav_data, rate=speed)
        if volume != 1.0:
            wav_data = wav_data * volume
        
        # 新增：计算音频处理耗时
        audio_process_cost = time.perf_counter() - audio_process_start_time
        print(f"✅ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 音频处理完成 - 耗时：{audio_process_cost:.4f} 秒")
        
        # ======================================
        # 新增：阶段4 - 音频保存，开始计时
        # ======================================
        save_start_time = time.perf_counter()
        print(f"⚙️ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 开始保存音频文件 - 路径：{save_path}")
        
        # 6. 保存音频
        sf.write(save_path, wav_data, sr)
        
        # 新增：计算音频保存耗时
        save_cost = time.perf_counter() - save_start_time
        print(f"✅ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 音频保存完成 - 耗时：{save_cost:.4f} 秒")
        
        # ======================================
        # 新增：阶段5 - 请求处理完成，计算总耗时
        # ======================================
        total_cost = time.perf_counter() - total_start_time
        print(f"🎉 [{time.strftime('%Y-%m-%d %H:%M:%S')}] 整个请求处理完成 - 总耗时：{total_cost:.4f} 秒")
        print(f"==================================================")
        
        # 7. 返回成功响应（新增：各阶段耗时数据）
        return {
            "code": 0,
            "message": "语音合成成功",
            "data": {
                "text": request.text,
                "spk_id": request.spk_id,
                "save_path": save_path,
                "sample_rate": sr,
                "speed": speed,
                "volume": volume,
                # 新增：耗时统计数据
                "time_statistics": {
                    "total_cost_seconds": round(total_cost, 4),
                    "model_infer_cost_seconds": round(infer_cost, 4),
                    "audio_process_cost_seconds": round(audio_process_cost, 4),
                    "audio_save_cost_seconds": round(save_cost, 4)
                }
            }
        }
    
    except FileNotFoundError as e:
        total_cost = time.perf_counter() - total_start_time
        print(f"❌ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 请求处理失败 - 文件不存在：{str(e)}")
        print(f"❌ 总耗时：{total_cost:.4f} 秒")
        print(f"==================================================")
        raise HTTPException(
            status_code=404,
            detail=f"文件不存在：{str(e)}（请检查参考音频或保存路径是否正确）"
        )
    except Exception as e:
        total_cost = time.perf_counter() - total_start_time
        print(f"❌ [{time.strftime('%Y-%m-%d %H:%M:%S')}] 请求处理失败 - 未知错误：{str(e)}")
        print(f"❌ 总耗时：{total_cost:.4f} 秒")
        print(f"==================================================")
        raise HTTPException(
            status_code=500,
            detail=f"语音合成失败：{str(e)}"
        )
